**Workflow file:** `.github/workflows/daily-pulse.yml`

```yaml
name: Daily Pulse
on:
  schedule:
    - cron: '0 4 * * *'  # 09:00 Tashkent time (UTC+5)
  workflow_dispatch:

jobs:
  pulse:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Run daily pulse
        env:
          API_URL: https://api.esli.app
          AGENT_EMAIL: ${{ secrets.AGENT_EMAIL }}
          AGENT_PASSWORD: ${{ secrets.AGENT_PASSWORD }}
          TELEGRAM_BOT_TOKEN: ${{ secrets.TELEGRAM_BOT_TOKEN }}
          TELEGRAM_CHAT_ID: ${{ secrets.TELEGRAM_CHAT_ID }}
        run: python3 agents/scripts/daily-pulse.py
      - name: Commit report
        run: |
          git config user.email "agent@esli.app"
          git config user.name "Esli Agent"
          git add agents/reports/daily-pulse/
          git diff --staged --quiet || git commit -m "chore: daily pulse $(date +%Y-%m-%d)"
          git push
```
